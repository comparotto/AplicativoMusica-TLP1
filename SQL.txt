create database aplicativomusica;

create table pessoa (
    id bigserial primary key,
    nome varchar(120) not null,
    idade integer not null
);

create table artista (
    id bigint primary key,
    nome_artistico varchar(120) not null,
    foreign key (id) references pessoa(id) on delete cascade
);

create table usuario (
    id bigint primary key,
    email varchar(150) not null,
    data_cadastro timestamp not null,
    foreign key (id) references pessoa(id) on delete cascade
);

create table banda (
    id bigserial primary key,
    nome varchar(120) not null,
    genero_musical varchar(80) not null
);

create table musica (
    id bigserial primary key,
    nome_musica varchar(150) not null,
    data_lancamento date,
    tempo_min real not null,
    genero_musical varchar(80) not null
);

create table playlist (
    id bigserial primary key,
    nome_playlist varchar(150) not null,
    duracao real not null default 0.0,
    usuario_id bigint not null,
    foreign key (usuario_id) references usuario(id) on delete cascade
);

create table playlist_musica (
    playlist_id bigint not null,
    musica_id bigint not null,
    posicao integer not null,
    primary key (playlist_id, musica_id),
    foreign key (playlist_id) references playlist(id) on delete cascade,
    foreign key (musica_id) references musica(id) on delete cascade
);
